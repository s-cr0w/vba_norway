Option Explicit

Sub ImportCSVToDatabase()

    Dim csvPath As Variant
    Dim csvWb As Workbook
    Dim csvWs As Worksheet
    Dim dbWs As Worksheet
    
    Dim lastCsvRow As Long
    Dim lastDbRow As Long
    Dim startRow As Long
    Dim numRows As Long
    Dim r As Long
    Dim dbRow As Long
    
    Dim kjoperLEI As String
    Dim selgerMortpartID As String
    Dim selgerFodelsedag As String
    
    Dim value3 As String
    Dim value16 As String
    Dim value7 As String
    Dim parts() As String
    Dim restValue As String
    Dim i As Long
    
    Dim fieldInfo() As Variant
    Dim maxColumns As Long
    
    Dim oldScreenUpdating As Boolean
    Dim oldDisplayAlerts As Boolean
    
    On Error GoTo ErrorHandler
    
    '------------------------------------------------------------
    ' Get the Database worksheet
    '------------------------------------------------------------
    Set dbWs = ThisWorkbook.Worksheets("Database")
    
    '------------------------------------------------------------
    ' Prompt for CSV file
    '------------------------------------------------------------
    csvPath = Application.GetOpenFilename( _
                FileFilter:="CSV Files (*.csv),*.csv", _
                Title:="Select CSV file")
    
    If csvPath = False Then
        Exit Sub
    End If
    
    '------------------------------------------------------------
    ' Prompt for additional values
    '------------------------------------------------------------
    kjoperLEI = InputBox("Kjoper LEI", "Kjoper LEI")
    
    If StrPtr(kjoperLEI) = 0 Then Exit Sub
    
    selgerMortpartID = InputBox("Selger Mortpart ID", "Selger Mortpart ID")
    
    If StrPtr(selgerMortpartID) = 0 Then Exit Sub
    
    selgerFodelsedag = InputBox( _
        "Selger Fodelsedag YYYY-MM-DD", _
        "Selger Fodelsedag")
    
    If StrPtr(selgerFodelsedag) = 0 Then Exit Sub
    
    '------------------------------------------------------------
    ' Improve performance
    '------------------------------------------------------------
    oldScreenUpdating = Application.ScreenUpdating
    oldDisplayAlerts = Application.DisplayAlerts
    
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    
    '------------------------------------------------------------
    ' Import CSV.
    '
    ' All first 16 columns are explicitly imported as TEXT.
    ' This prevents Excel from converting dates/numbers automatically.
    '------------------------------------------------------------
    
    ReDim fieldInfo(1 To 16)
    
    For i = 1 To 16
        fieldInfo(i) = Array(i, xlTextFormat)
    Next i
    
    Workbooks.OpenText _
        Filename:=CStr(csvPath), _
        Origin:=65001, _
        StartRow:=1, _
        DataType:=xlDelimited, _
        TextQualifier:=xlTextQualifierDoubleQuote, _
        ConsecutiveDelimiter:=False, _
        Tab:=False, _
        Semicolon:=False, _
        Comma:=True, _
        Space:=False, _
        Other:=False, _
        FieldInfo:=fieldInfo
    
    Set csvWb = ActiveWorkbook
    Set csvWs = csvWb.Worksheets(1)
    
    '------------------------------------------------------------
    ' Find last row in CSV.
    '
    ' CSV has NO header row.
    '------------------------------------------------------------
    lastCsvRow = csvWs.Cells(csvWs.Rows.Count, 2).End(xlUp).Row
    
    If lastCsvRow < 1 Then
        MsgBox "The CSV file contains no data.", vbExclamation
        GoTo CleanUp
    End If
    
    numRows = lastCsvRow
    
    '------------------------------------------------------------
    ' Find first available row in Database.
    '
    ' Row 1 is reserved for headers.
    ' Existing data is never overwritten.
    '------------------------------------------------------------
    lastDbRow = dbWs.Cells(dbWs.Rows.Count, "F").End(xlUp).Row
    
    If lastDbRow < 2 Then
        startRow = 2
    Else
        startRow = lastDbRow + 1
    End If
    
    '------------------------------------------------------------
    ' Import each CSV row
    '------------------------------------------------------------
    For r = 1 To lastCsvRow
        
        dbRow = startRow + (r - 1)
        
        '========================================================
        ' CSV column 2 -> Database F
        '========================================================
        dbWs.Cells(dbRow, "F").Value2 = CStr(csvWs.Cells(r, 2).Value2)
        
        '========================================================
        ' CSV column 4 -> Database S and T
        '========================================================
        dbWs.Cells(dbRow, "S").Value2 = CStr(csvWs.Cells(r, 4).Value2)
        dbWs.Cells(dbRow, "T").Value2 = CStr(csvWs.Cells(r, 4).Value2)
        
        '========================================================
        ' CSV column 7 -> Database AG
        '
        ' Convert to string and remove ALL leading "-".
        '========================================================
        value7 = CStr(csvWs.Cells(r, 7).Value2)
        
        Do While Len(value7) > 0 And Left$(value7, 1) = "-"
            value7 = Mid$(value7, 2)
        Loop
        
        dbWs.Cells(dbRow, "AG").NumberFormat = "@"
        dbWs.Cells(dbRow, "AG").Value2 = value7
        
        '========================================================
        ' CSV column 3 -> Database AL
        '
        ' Expected CSV format:
        '
        ' 2026-06-23 14:12:56.214000
        '
        ' Output:
        '
        ' 2026-06-23T14:12:56.21
        '
        ' The result is explicitly stored as TEXT.
        '========================================================
        
        value3 = CStr(csvWs.Cells(r, 3).Value2)
        
        ' Remove any leading/trailing whitespace
        value3 = Trim$(value3)
        
        If Len(value3) >= 22 Then
            
            ' Build the output explicitly rather than using
            ' date/time conversion functions.
            '
            ' Characters:
            ' 1-10  = YYYY-MM-DD
            ' 12-19 = HH:MM:SS
            ' 21-22 = hundredths of a second
            
            value3 = _
                Mid$(value3, 1, 10) & _
                "T" & _
                Mid$(value3, 12, 8) & _
                Mid$(value3, 20, 3)
            
            ' The above produces:
            '
            ' 2026-06-23T14:12:56.21
            '
            ' Store as text.
            dbWs.Cells(dbRow, "AL").NumberFormat = "@"
            dbWs.Cells(dbRow, "AL").Value2 = value3
            
        Else
            
            dbWs.Cells(dbRow, "AL").NumberFormat = "@"
            dbWs.Cells(dbRow, "AL").Value2 = ""
            
        End If
        
        '========================================================
        ' CSV column 16
        '
        ' Split on spaces.
        ' Last value -> EC
        ' Everything before last value -> ED
        '========================================================
        value16 = Trim$(CStr(csvWs.Cells(r, 16).Value2))
        
        If Len(value16) > 0 Then
            
            parts = Split(value16, " ")
            
            ' Last value -> EC
            dbWs.Cells(dbRow, "EC").Value2 = parts(UBound(parts))
            
            ' Everything before last value -> ED
            restValue = ""
            
            If UBound(parts) > 0 Then
                
                For i = LBound(parts) To UBound(parts) - 1
                    
                    If restValue <> "" Then
                        restValue = restValue & " "
                    End If
                    
                    restValue = restValue & parts(i)
                    
                Next i
                
            End If
            
            dbWs.Cells(dbRow, "ED").Value2 = restValue
            
        Else
            
            dbWs.Cells(dbRow, "EC").Value2 = ""
            dbWs.Cells(dbRow, "ED").Value2 = ""
            
        End If
        
        '========================================================
        ' Prompted values
        '========================================================
        
        ' Kjoper LEI
        dbWs.Cells(dbRow, "DB").Value2 = kjoperLEI
        
        ' Selger Mortpart ID
        dbWs.Cells(dbRow, "DD").Value2 = selgerMortpartID
        
        ' Selger Fodelsedag
        '
        ' Keep the cell formatted as GENERAL but force Excel
        ' to retain the YYYY-MM-DD value as text.
        '
        ' The leading apostrophe is not displayed in Excel.
        dbWs.Cells(dbRow, "EE").NumberFormat = "General"
        dbWs.Cells(dbRow, "EE").Value2 = "'" & selgerFodelsedag
        
        '========================================================
        ' Fixed values
        '========================================================
        
        ' Column A
        '
        ' Store as TEXT so the decimal separator is always ".".
        dbWs.Cells(dbRow, "A").NumberFormat = "@"
        dbWs.Cells(dbRow, "A").Value2 = "2.4"
        
        dbWs.Cells(dbRow, "C").Value2 = "EQ"
        dbWs.Cells(dbRow, "D").Value2 = "CS"
        dbWs.Cells(dbRow, "E").Value2 = "ISIN"
        
        dbWs.Cells(dbRow, "W").Value2 = "MIC"
        dbWs.Cells(dbRow, "X").Value2 = "XOFF"
        dbWs.Cells(dbRow, "AA").Value2 = "NA"
        dbWs.Cells(dbRow, "AB").Value2 = "NOK"
        
        dbWs.Cells(dbRow, "CW").Value2 = "N"
        dbWs.Cells(dbRow, "CY").Value2 = "N"
        
        dbWs.Cells(dbRow, "DA").Value2 = "LEI"
        dbWs.Cells(dbRow, "DC").Value2 = "NATIONAL_ID_NUM"
        
        dbWs.Cells(dbRow, "DI").Value2 = "LEI"
        dbWs.Cells(dbRow, "DJ").Value2 = kjoperLEI
        dbWs.Cells(dbRow, "DK").Value2 = "Y"
        
        dbWs.Cells(dbRow, "DL").Value2 = "LEI"
        dbWs.Cells(dbRow, "DM").Value2 = kjoperLEI
        
        dbWs.Cells(dbRow, "DN").Value2 = "LEI"
        dbWs.Cells(dbRow, "DO").Value2 = kjoperLEI
        
        dbWs.Cells(dbRow, "DP").Value2 = "NO"
        
        dbWs.Cells(dbRow, "DZ").Value2 = "NATIONAL_ID_NUM"
        dbWs.Cells(dbRow, "EA").Value2 = selgerMortpartID
        dbWs.Cells(dbRow, "EB").Value2 = "NO"
        
        dbWs.Cells(dbRow, "EP").Value2 = "NOT_WITHIN_FIRM"
        dbWs.Cells(dbRow, "EQ").Value2 = "NORE"
        
        dbWs.Cells(dbRow, "FA").Value2 = "AOTC"
        dbWs.Cells(dbRow, "FO").Value2 = "N"
        
    Next r
    
    '------------------------------------------------------------
    ' Close CSV without saving
    '------------------------------------------------------------
    csvWb.Close SaveChanges:=False
    Set csvWb = Nothing
    
    Application.ScreenUpdating = oldScreenUpdating
    Application.DisplayAlerts = oldDisplayAlerts
    
    MsgBox numRows & " rows imported successfully." & vbCrLf & _
           "Starting row: " & startRow & vbCrLf & _
           "Ending row: " & (startRow + numRows - 1), _
           vbInformation, "Import complete"
    
    Exit Sub

CleanUp:

    If Not csvWb Is Nothing Then
        csvWb.Close SaveChanges:=False
    End If
    
    Application.ScreenUpdating = oldScreenUpdating
    Application.DisplayAlerts = oldDisplayAlerts
    
    Exit Sub

ErrorHandler:

    If Not csvWb Is Nothing Then
        csvWb.Close SaveChanges:=False
    End If
    
    Application.ScreenUpdating = oldScreenUpdating
    Application.DisplayAlerts = oldDisplayAlerts
    
    MsgBox "An error occurred:" & vbCrLf & vbCrLf & _
           Err.Number & " - " & Err.Description, _
           vbCritical, "Import failed"

End Sub
