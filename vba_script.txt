Option Explicit

Sub ImportCSVToDatabase()

    Dim csvPath As Variant
    Dim dbWs As Worksheet
    
    Dim csvLines As Collection
    Dim fields As Variant
    
    Dim lastDbRow As Long
    Dim startRow As Long
    Dim numRows As Long
    Dim r As Long
    Dim dbRow As Long
    
    Dim kjoperLEI As String
    Dim selgerMortpartID As String
    Dim selgerFodelsedag As String
    
    Dim value3 As String
    Dim value7 As String
    Dim value16 As String
    
    Dim parts() As String
    Dim restValue As String
    Dim i As Long
    
    Dim oldScreenUpdating As Boolean
    Dim oldDisplayAlerts As Boolean
    
    On Error GoTo ErrorHandler
    
    '------------------------------------------------------------
    ' Get Database worksheet
    '------------------------------------------------------------
    Set dbWs = ThisWorkbook.Worksheets("Database")
    
    '------------------------------------------------------------
    ' Prompt for CSV file
    '------------------------------------------------------------
    csvPath = Application.GetOpenFilename( _
                FileFilter:="CSV Files (*.csv),*.csv", _
                Title:="Select CSV file")
    
    If csvPath = False Then
        Exit Sub
    End If
    
    '------------------------------------------------------------
    ' Prompt for additional values
    '------------------------------------------------------------
    kjoperLEI = InputBox("Kjoper LEI", "Kjoper LEI")
    
    If StrPtr(kjoperLEI) = 0 Then Exit Sub
    
    selgerMortpartID = InputBox( _
        "Selger Mortpart ID", _
        "Selger Mortpart ID")
    
    If StrPtr(selgerMortpartID) = 0 Then Exit Sub
    
    selgerFodelsedag = InputBox( _
        "Selger Fodelsedag YYYY-MM-DD", _
        "Selger Fodelsedag")
    
    If StrPtr(selgerFodelsedag) = 0 Then Exit Sub
    
    '------------------------------------------------------------
    ' Improve performance
    '------------------------------------------------------------
    oldScreenUpdating = Application.ScreenUpdating
    oldDisplayAlerts = Application.DisplayAlerts
    
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    
    '------------------------------------------------------------
    ' Read CSV as RAW TEXT.
    '
    ' Excel is NOT used to open the CSV.
    ' This prevents automatic date/number conversion.
    '------------------------------------------------------------
    Set csvLines = ReadCSVFile(CStr(csvPath))
    
    If csvLines.Count = 0 Then
        MsgBox "The CSV file contains no data.", vbExclamation
        GoTo CleanUp
    End If
    
    numRows = csvLines.Count
    
    '------------------------------------------------------------
    ' Find first available row in Database.
    '
    ' Row 1 is reserved for headers.
    ' Existing data is never overwritten.
    '------------------------------------------------------------
    lastDbRow = dbWs.Cells(dbWs.Rows.Count, "F").End(xlUp).Row
    
    If lastDbRow < 2 Then
        startRow = 2
    Else
        startRow = lastDbRow + 1
    End If
    
    '------------------------------------------------------------
    ' Import each CSV row
    '------------------------------------------------------------
    For r = 1 To csvLines.Count
        
        dbRow = startRow + (r - 1)
        
        '--------------------------------------------------------
        ' Parse CSV line
        '--------------------------------------------------------
        fields = ParseCSVLine(CStr(csvLines(r)))
        
        ' Make sure CSV contains at least 16 columns
        If UBound(fields) < 15 Then
            
            MsgBox "CSV row " & r & _
                   " contains fewer than 16 columns." & vbCrLf & _
                   "Import stopped.", _
                   vbCritical, "Invalid CSV"
            
            GoTo CleanUp
            
        End If
        
        '========================================================
        ' CSV column 2 -> Database F
        '========================================================
        dbWs.Cells(dbRow, "F").Value2 = CStr(fields(1))
        
        '========================================================
        ' CSV column 4 -> Database S and T
        '========================================================
        dbWs.Cells(dbRow, "S").Value2 = CStr(fields(3))
        dbWs.Cells(dbRow, "T").Value2 = CStr(fields(3))
        
        '========================================================
        ' CSV column 7 -> Database AG
        '
        ' Remove ALL leading "-" characters.
        '========================================================
        value7 = CStr(fields(6))
        
        Do While Len(value7) > 0 And Left$(value7, 1) = "-"
            value7 = Mid$(value7, 2)
        Loop
        
        dbWs.Cells(dbRow, "AG").NumberFormat = "@"
        dbWs.Cells(dbRow, "AG").Value2 = value7
        
        '========================================================
        ' CSV column 3 -> Database AL
        '
        ' Input:
        '
        ' 2026-06-23 14:12:56.214000
        '
        ' Output:
        '
        ' 2026-06-23T14:12:56
        '
        ' Fractional seconds are discarded.
        '========================================================
        value3 = Trim$(CStr(fields(2)))
        
        If Len(value3) >= 19 Then
            
            value3 = _
                Mid$(value3, 1, 10) & _
                "T" & _
                Mid$(value3, 12, 8)
            
        Else
            
            value3 = ""
            
        End If
        
        ' Force AL to TEXT
        dbWs.Cells(dbRow, "AL").NumberFormat = "@"
        dbWs.Cells(dbRow, "AL").Value2 = value3
        
        '========================================================
        ' CSV column 16 -> EC and ED
        '
        ' Split on spaces.
        ' Last value -> EC
        ' Everything before last value -> ED
        '========================================================
        value16 = Trim$(CStr(fields(15)))
        
        If Len(value16) > 0 Then
            
            parts = Split(value16, " ")
            
            ' Last value -> EC
            dbWs.Cells(dbRow, "EC").Value2 = parts(UBound(parts))
            
            ' Everything before last value -> ED
            restValue = ""
            
            If UBound(parts) > 0 Then
                
                For i = LBound(parts) To UBound(parts) - 1
                    
                    If restValue <> "" Then
                        restValue = restValue & " "
                    End If
                    
                    restValue = restValue & parts(i)
                    
                Next i
                
            End If
            
            dbWs.Cells(dbRow, "ED").Value2 = restValue
            
        Else
            
            dbWs.Cells(dbRow, "EC").Value2 = ""
            dbWs.Cells(dbRow, "ED").Value2 = ""
            
        End If
        
        '========================================================
        ' Prompted values
        '========================================================
        
        dbWs.Cells(dbRow, "DB").Value2 = kjoperLEI
        dbWs.Cells(dbRow, "DD").Value2 = selgerMortpartID
        
        '--------------------------------------------------------
        ' EE
        '
        ' Keep Excel number format as General but force the
        ' value to remain text instead of becoming a date.
        '--------------------------------------------------------
        dbWs.Cells(dbRow, "EE").NumberFormat = "General"
        dbWs.Cells(dbRow, "EE").Value2 = "'" & selgerFodelsedag
        
        '========================================================
        ' Fixed values
        '========================================================
        
        ' A
        ' Stored as text to guarantee decimal ".".
        dbWs.Cells(dbRow, "A").NumberFormat = "@"
        dbWs.Cells(dbRow, "A").Value2 = "2.4"
        
        dbWs.Cells(dbRow, "C").Value2 = "EQ"
        dbWs.Cells(dbRow, "D").Value2 = "CS"
        dbWs.Cells(dbRow, "E").Value2 = "ISIN"
        
        dbWs.Cells(dbRow, "W").Value2 = "MIC"
        dbWs.Cells(dbRow, "X").Value2 = "XOFF"
        dbWs.Cells(dbRow, "AA").Value2 = "NA"
        dbWs.Cells(dbRow, "AB").Value2 = "NOK"
        
        dbWs.Cells(dbRow, "CW").Value2 = "N"
        dbWs.Cells(dbRow, "CY").Value2 = "N"
        
        dbWs.Cells(dbRow, "DA").Value2 = "LEI"
        dbWs.Cells(dbRow, "DC").Value2 = "NATIONAL_ID_NUM"
        
        dbWs.Cells(dbRow, "DI").Value2 = "LEI"
        dbWs.Cells(dbRow, "DJ").Value2 = kjoperLEI
        dbWs.Cells(dbRow, "DK").Value2 = "Y"
        
        dbWs.Cells(dbRow, "DL").Value2 = "LEI"
        dbWs.Cells(dbRow, "DM").Value2 = kjoperLEI
        
        dbWs.Cells(dbRow, "DN").Value2 = "LEI"
        dbWs.Cells(dbRow, "DO").Value2 = kjoperLEI
        
        dbWs.Cells(dbRow, "DP").Value2 = "NO"
        
        dbWs.Cells(dbRow, "DZ").Value2 = "NATIONAL_ID_NUM"
        dbWs.Cells(dbRow, "EA").Value2 = selgerMortpartID
        dbWs.Cells(dbRow, "EB").Value2 = "NO"
        
        dbWs.Cells(dbRow, "EP").Value2 = "NOT_WITHIN_FIRM"
        dbWs.Cells(dbRow, "EQ").Value2 = "NORE"
        
        dbWs.Cells(dbRow, "FA").Value2 = "AOTC"
        dbWs.Cells(dbRow, "FO").Value2 = "N"
        
    Next r
    
    '------------------------------------------------------------
    ' Restore Excel settings
    '------------------------------------------------------------
    Application.ScreenUpdating = oldScreenUpdating
    Application.DisplayAlerts = oldDisplayAlerts
    
    MsgBox numRows & " rows imported successfully." & vbCrLf & _
           "Starting row: " & startRow & vbCrLf & _
           "Ending row: " & (startRow + numRows - 1), _
           vbInformation, "Import complete"
    
    Exit Sub

CleanUp:

    Application.ScreenUpdating = oldScreenUpdating
    Application.DisplayAlerts = oldDisplayAlerts
    
    Exit Sub

ErrorHandler:

    Application.ScreenUpdating = oldScreenUpdating
    Application.DisplayAlerts = oldDisplayAlerts
    
    MsgBox "An error occurred:" & vbCrLf & vbCrLf & _
           Err.Number & " - " & Err.Description, _
           vbCritical, "Import failed"

End Sub


'================================================================
' READ CSV FILE
'
' Determines whether the file has a UTF-8 BOM.
'
' UTF-8 with BOM:
'   Decode as UTF-8.
'
' No BOM:
'   Treat as Windows/ANSI encoded.
'
' This is important for Norwegian/Danish characters such as:
'
' æ Æ
' ø Ø
' å Å
'================================================================
Function ReadCSVFile(ByVal filePath As String) As Collection

    Dim fileNumber As Integer
    Dim fileLength As Long
    Dim fileData() As Byte
    Dim fileText As String
    
    Dim lines As Variant
    Dim result As New Collection
    Dim i As Long
    
    fileNumber = FreeFile
    
    '------------------------------------------------------------
    ' Open file as binary
    '------------------------------------------------------------
    Open filePath For Binary Access Read As #fileNumber
    
    fileLength = LOF(fileNumber)
    
    If fileLength = 0 Then
        
        Close #fileNumber
        
        Set ReadCSVFile = result
        
        Exit Function
        
    End If
    
    ReDim fileData(0 To fileLength - 1)
    
    Get #fileNumber, , fileData
    
    Close #fileNumber
    
    '------------------------------------------------------------
    ' Check for UTF-8 BOM
    '
    ' UTF-8 BOM:
    ' EF BB BF
    '------------------------------------------------------------
    If fileLength >= 3 Then
        
        If fileData(0) = &HEF And _
           fileData(1) = &HBB And _
           fileData(2) = &HBF Then
            
            '----------------------------------------------------
            ' UTF-8 with BOM
            '----------------------------------------------------
            fileText = UTF8BytesToString(fileData, 3)
            
        Else
            
            '----------------------------------------------------
            ' No BOM.
            '
            ' Read using the Windows/ANSI code page.
            ' This preserves Nordic characters such as:
            ' æ ø å Æ Ø Å
            '----------------------------------------------------
            fileText = ANSIBytesToString(fileData)
            
        End If
        
    Else
        
        '--------------------------------------------------------
        ' Very small file / no BOM
        '--------------------------------------------------------
        fileText = ANSIBytesToString(fileData)
        
    End If
    
    '------------------------------------------------------------
    ' Normalize line endings
    '------------------------------------------------------------
    fileText = Replace(fileText, vbCrLf, vbLf)
    fileText = Replace(fileText, vbCr, vbLf)
    
    lines = Split(fileText, vbLf)
    
    For i = LBound(lines) To UBound(lines)
        
        ' Ignore empty lines
        If Len(lines(i)) > 0 Then
            result.Add lines(i)
        End If
        
    Next i
    
    Set ReadCSVFile = result

End Function


'================================================================
' CONVERT ANSI BYTES TO VBA STRING
'
' Uses the Windows system code page.
'
' On a Nordic Windows installation this correctly handles:
'
' æ Æ
' ø Ø
' å Å
'================================================================
Function ANSIBytesToString(ByRef bytes() As Byte) As String

    Dim tempFile As Integer
    Dim tempPath As String
    Dim fileLength As Long
    Dim result As String
    
    '------------------------------------------------------------
    ' Use a temporary file so Windows itself performs the
    ' correct ANSI -> Unicode conversion.
    '------------------------------------------------------------
    tempPath = Environ$("TEMP") & "\vba_csv_temp_" & _
               Format$(Timer * 1000, "0") & ".tmp"
    
    tempFile = FreeFile
    
    Open tempPath For Binary Access Write As #tempFile
    
    Put #tempFile, , bytes
    
    Close #tempFile
    
    '------------------------------------------------------------
    ' Open the temporary file as normal text.
    '
    ' VBA will use the Windows system ANSI code page.
    '------------------------------------------------------------
    tempFile = FreeFile
    
    Open tempPath For Input As #tempFile
    
    fileLength = LOF(tempFile)
    
    If fileLength > 0 Then
        result = Input$(fileLength, #tempFile)
    Else
        result = ""
    End If
    
    Close #tempFile
    
    '------------------------------------------------------------
    ' Delete temporary file
    '------------------------------------------------------------
    On Error Resume Next
    Kill tempPath
    On Error GoTo 0
    
    ANSIBytesToString = result

End Function


'================================================================
' UTF-8 BYTE ARRAY -> VBA STRING
'
' Used only when the CSV has a UTF-8 BOM.
'================================================================
Function UTF8BytesToString( _
    ByRef bytes() As Byte, _
    ByVal startPosition As Long) As String

    Dim result As String
    Dim i As Long
    Dim b1 As Long
    Dim b2 As Long
    Dim b3 As Long
    Dim b4 As Long
    Dim codePoint As Long
    
    result = ""
    
    i = startPosition
    
    Do While i <= UBound(bytes)
        
        b1 = bytes(i)
        
        '--------------------------------------------------------
        ' Standard ASCII
        '--------------------------------------------------------
        If b1 < &H80 Then
            
            result = result & ChrW$(b1)
            i = i + 1
        
        '--------------------------------------------------------
        ' Two-byte UTF-8
        '--------------------------------------------------------
        ElseIf b1 >= &HC0 And b1 <= &HDF Then
            
            If i + 1 <= UBound(bytes) Then
                
                b2 = bytes(i + 1)
                
                codePoint = _
                    ((b1 And &H1F) * &H40) + _
                    (b2 And &H3F)
                
                result = result & ChrW$(codePoint)
                
                i = i + 2
                
            Else
                
                i = i + 1
                
            End If
        
        '--------------------------------------------------------
        ' Three-byte UTF-8
        '--------------------------------------------------------
        ElseIf b1 >= &HE0 And b1 <= &HEF Then
            
            If i + 2 <= UBound(bytes) Then
                
                b2 = bytes(i + 1)
                b3 = bytes(i + 2)
                
                codePoint = _
                    ((b1 And &HF) * &H1000) + _
                    ((b2 And &H3F) * &H40) + _
                    (b3 And &H3F)
                
                result = result & ChrW$(codePoint)
                
                i = i + 3
                
            Else
                
                i = i + 1
                
            End If
        
        '--------------------------------------------------------
        ' Four-byte UTF-8
        '--------------------------------------------------------
        ElseIf b1 >= &HF0 And b1 <= &HF7 Then
            
            If i + 3 <= UBound(bytes) Then
                
                b2 = bytes(i + 1)
                b3 = bytes(i + 2)
                b4 = bytes(i + 3)
                
                codePoint = _
                    ((b1 And &H7) * &H40000) + _
                    ((b2 And &H3F) * &H1000) + _
                    ((b3 And &H3F) * &H40) + _
                    (b4 And &H3F)
                
                ' Convert to UTF-16 surrogate pair
                If codePoint >= &H10000 Then
                    
                    codePoint = codePoint - &H10000
                    
                    result = result & _
                        ChrW$(&HD800 Or (codePoint \ &H400)) & _
                        ChrW$(&HDC00 Or (codePoint And &H3FF))
                    
                Else
                    
                    result = result & ChrW$(codePoint)
                    
                End If
                
                i = i + 4
                
            Else
                
                i = i + 1
                
            End If
        
        Else
            
            ' Invalid UTF-8 byte; skip it
            i = i + 1
            
        End If
        
    Loop
    
    UTF8BytesToString = result

End Function


'================================================================
' PARSE CSV LINE
'
' Handles:
'   - Commas
'   - Quoted fields
'   - Commas inside quoted fields
'   - Escaped quotes ("")
'================================================================
Function ParseCSVLine(ByVal line As String) As Variant

    Dim result() As String
    Dim field As String
    
    Dim i As Long
    Dim fieldIndex As Long
    Dim inQuotes As Boolean
    Dim currentChar As String
    Dim nextChar As String
    
    ReDim result(0 To 0)
    
    fieldIndex = 0
    field = ""
    inQuotes = False
    
    For i = 1 To Len(line)
        
        currentChar = Mid$(line, i, 1)
        
        If currentChar = """" Then
            
            If inQuotes Then
                
                '------------------------------------------------
                ' Escaped quote ""
                '------------------------------------------------
                If i < Len(line) Then
                    
                    nextChar = Mid$(line, i + 1, 1)
                    
                    If nextChar = """" Then
                        
                        field = field & """"
                        i = i + 1
                        
                    Else
                        
                        inQuotes = False
                        
                    End If
                    
                Else
                    
                    inQuotes = False
                    
                End If
                
            Else
                
                inQuotes = True
                
            End If
            
        ElseIf currentChar = "," And Not inQuotes Then
            
            result(fieldIndex) = field
            
            fieldIndex = fieldIndex + 1
            
            ReDim Preserve result(0 To fieldIndex)
            
            field = ""
            
        Else
            
            field = field & currentChar
            
        End If
        
    Next i
    
    '------------------------------------------------------------
    ' Add final field
    '------------------------------------------------------------
    result(fieldIndex) = field
    
    ParseCSVLine = result

End Function
